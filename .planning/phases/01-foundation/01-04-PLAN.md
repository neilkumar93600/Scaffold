    Create the directory .github/workflows/ if it does not exist, then create ci.yml:

    ```yaml
    name: CI

    on:
      pull_request:
      push:
        branches: [main]
      schedule:
        - cron: '0 2 * * *'  # nightly at 2am UTC

    jobs:
      ci:
        name: Lint, Typecheck, Test, Build
        runs-on: ubuntu-latest

        steps:
          - name: Checkout
            uses: actions/checkout@v4

          - name: Install pnpm
            uses: pnpm/action-setup@v4
            with:
              version: 9

          - name: Setup Node.js
            uses: actions/setup-node@v4
            with:
              node-version: '22'
              cache: 'pnpm'

          - name: Install dependencies
            run: pnpm install --frozen-lockfile

          - name: Lint
            # Uses eslint directly — next lint is removed in Next.js 16
            run: pnpm lint

          - name: Typecheck
            # Generate Next.js type declarations before tsc runs
            run: pnpm exec next build --no-lint 2>/dev/null || pnpm exec next typegen 2>/dev/null || true
            continue-on-error: true
            name: Generate Next.js types (best-effort)

          - name: Typecheck
            run: pnpm typecheck

          - name: Test
            # --run flag is required in CI — without it Vitest enters watch mode and hangs
            run: pnpm test --run

          - name: Build
            run: pnpm build
            env:
              # Stub values so next build does not fail on missing real env vars at build time.
              # These are NOT real credentials — safe to have in the workflow file.
              NEXT_PUBLIC_SUPABASE_URL: 'https://stub.supabase.co'
              NEXT_PUBLIC_SUPABASE_ANON_KEY: 'stub-anon-key'
              SUPABASE_SERVICE_ROLE_KEY: 'stub-service-role-key'
              DATABASE_URL: 'postgresql://stub:stub@localhost:5432/stub'
              NEXT_PUBLIC_SENTRY_DSN: 'https://stub@stub.ingest.sentry.io/0'

    ```
